import express from "express";
import cors from "cors";
import multer from "multer";
import { randomUUID } from "node:crypto";
import "dotenv/config";

import { chat, extractJson } from "./llm.js";
import { extractText } from "./extract-text.js";
import { fetchNews, formatNews } from "./news.js";
import { DASHBOARD_SCHEMA, EXTRACTION_SYSTEM, FIELDS, REPORT_SYSTEM, CHAT_SYSTEM } from "./prompts.js";

const app = express();
app.use(cors({ origin: process.env.FRONTEND_ORIGIN }));
app.use(express.json({ limit: "2mb" }));
const upload = multer({ storage: multer.memoryStorage(), limits: { fileSize: 20 * 1024 * 1024 } });

// In-memory store for the demo. Replace with PostgreSQL/MongoDB for production.
const store = new Map();

// Keep prompts within a safe size. For big documents, replace this with chunking + embeddings.
const MAX_CHARS = Number(process.env.MAX_DOC_CHARS) || 24000;
const clip = (t) => (t.length > MAX_CHARS ? t.slice(0, MAX_CHARS) : t);

// 1) Upload a document -> text -> dashboard JSON
app.post("/api/documents", upload.single("file"), async (req, res) => {
  try {
    if (!req.file) return res.status(400).json({ error: "No file uploaded" });

    const text = await extractText(req.file);
    if (text.trim().length < 50) {
      return res.status(422).json({ error: "Little or no text found. The file may be a scan (OCR needed)." });
    }

    const dashboard = await extractJson(
      EXTRACTION_SYSTEM,
      `Schema:\n${JSON.stringify(DASHBOARD_SCHEMA, null, 2)}\n\nDocument:\n${clip(text)}`
    );

    const id = randomUUID();
    store.set(id, { id, filename: req.file.originalname, text, dashboard, reports: {}, news: [] });
    res.json({ id, filename: req.file.originalname, dashboard });
  } catch (err) {
    console.error(err);
    res.status(500).json({ error: err.message });
  }
});

// 2) Generate the 5 reports (in parallel) + market news context
app.post("/api/documents/:id/reports", async (req, res) => {
  try {
    const doc = store.get(req.params.id);
    if (!doc) return res.status(404).json({ error: "Document not found" });

    const industry = req.body?.industry || "beverage industry";
    const news = await fetchNews(`${industry} supply chain OR sales OR market`);
    doc.news = news;
    const newsText = formatNews(news);

    const entries = await Promise.all(
      Object.entries(FIELDS).map(async ([key, field]) => {
        const report = await chat([
          { role: "system", content: REPORT_SYSTEM },
          {
            role: "user",
            content:
              `Write the "${field.title}" report.\nFocus areas: ${field.focus}\n\n` +
              `Department KPIs (JSON):\n${JSON.stringify(doc.dashboard.departments?.[key] ?? {})}\n\n` +
              `Cross-department KPIs (JSON):\n${JSON.stringify(doc.dashboard.cross_department ?? {})}\n\n` +
              `Document text:\n${clip(doc.text)}\n\n` +
              `Recent news (market context):\n${newsText}`
          }
        ]);
        return [key, report];
      })
    );

    doc.reports = Object.fromEntries(entries);
    res.json({ reports: doc.reports, news });
  } catch (err) {
    console.error(err);
    res.status(500).json({ error: err.message });
  }
});

// 3) Chatbot: answers from the document, the reports and the news
app.post("/api/documents/:id/chat", async (req, res) => {
  try {
    const doc = store.get(req.params.id);
    if (!doc) return res.status(404).json({ error: "Document not found" });

    const { message, history = [] } = req.body;
    if (!message) return res.status(400).json({ error: "Missing message" });

    const context =
      `DOCUMENT: ${doc.filename}\n\nEXTRACTED DATA:\n${JSON.stringify(doc.dashboard)}\n\n` +
      `REPORTS:\n${Object.entries(doc.reports).map(([k, v]) => `## ${FIELDS[k].title}\n${v}`).join("\n\n")}\n\n` +
      `MARKET NEWS:\n${formatNews(doc.news)}\n\n` +
      `DOCUMENT TEXT:\n${clip(doc.text)}`;

    const answer = await chat([
      { role: "system", content: `${CHAT_SYSTEM}\n\nCONTEXT:\n${context}` },
      ...history.slice(-10),
      { role: "user", content: message }
    ], { temperature: 0.3 });

    res.json({ answer });
  } catch (err) {
    console.error(err);
    res.status(500).json({ error: err.message });
  }
});

app.listen(process.env.PORT || 3001, () => console.log("API running on port", process.env.PORT || 3001));


===== backend\llm.js =====

import OpenAI from "openai";
import "dotenv/config";

// Works with any OpenAI-compatible provider (AI Grid, Together, DeepInfra, OpenRouter, Ollama, vLLM...).
// Everything is configured in .env, no code change needed.
const { LLM_BASE_URL, LLM_API_KEY, LLM_MODEL } = process.env;

if (!LLM_BASE_URL || !LLM_MODEL) {
  throw new Error("Missing LLM_BASE_URL or LLM_MODEL in backend/.env (see .env.example).");
}

const client = new OpenAI({
  apiKey: LLM_API_KEY || "not-needed",   // some local servers need no key
  baseURL: LLM_BASE_URL
});

// Gemma-friendly switches
const JSON_MODE = process.env.LLM_JSON_MODE !== "false";       // ask the API for JSON output
const MERGE_SYSTEM = process.env.LLM_MERGE_SYSTEM === "true";  // Gemma templates often have no "system" role

// If the provider has no system role, fold the system text into the first user message.
function prepare(messages) {
  if (!MERGE_SYSTEM) return messages;
  const system = messages.filter((m) => m.role === "system").map((m) => m.content).join("\n\n");
  const rest = messages.filter((m) => m.role !== "system");
  if (!system) return rest;
  const i = rest.findIndex((m) => m.role === "user");
  if (i === -1) return [{ role: "user", content: system }, ...rest];
  rest[i] = { ...rest[i], content: `${system}\n\n---\n\n${rest[i].content}` };
  return rest;
}

export async function chat(messages, { json = false, temperature = 0.2 } = {}) {
  const base = { model: LLM_MODEL, messages: prepare(messages), temperature };

  try {
    const res = await client.chat.completions.create({
      ...base,
      ...(json && JSON_MODE ? { response_format: { type: "json_object" } } : {})
    });
    return res.choices[0].message.content;
  } catch (err) {
    // Some providers reject response_format for Gemma: retry once without it.
    if (json && JSON_MODE && err.status === 400) {
      console.warn("Provider rejected JSON mode, retrying without it. Set LLM_JSON_MODE=false to skip this.");
      const res = await client.chat.completions.create(base);
      return res.choices[0].message.content;
    }
    throw err;
  }
}

// Pulls a JSON object out of the reply even if the model adds text or ```json fences.
function parseJson(raw) {
  try {
    return JSON.parse(raw);
  } catch {
    const start = raw.indexOf("{");
    const end = raw.lastIndexOf("}");
    if (start === -1 || end === -1) throw new Error("Model did not return JSON.");
    return JSON.parse(raw.slice(start, end + 1));
  }
}

export async function extractJson(system, user) {
  const attempt = async (extra = "") =>
    parseJson(
      await chat(
        [
          { role: "system", content: system },
          { role: "user", content: user + extra }
        ],
        { json: true, temperature: 0 }
      )
    );

  try {
    return await attempt();
  } catch {
    // One retry with a stricter reminder (small models sometimes break the JSON).
    return await attempt("\n\nReturn ONLY one valid JSON object. No text before or after it.");
  }
}

===== backend\news.js =====

import "dotenv/config";

// Fetches recent headlines to give the LLM market context.
// Uses NewsAPI's "everything" endpoint; swap for GNews or another provider if you prefer.
export async function fetchNews(query, limit = 8) {
  if (!process.env.NEWS_API_KEY) return [];

  const url = new URL("https://newsapi.org/v2/everything");
  url.searchParams.set("q", query);
  url.searchParams.set("language", "en");
  url.searchParams.set("sortBy", "publishedAt");
  url.searchParams.set("pageSize", String(limit));
  url.searchParams.set("apiKey", process.env.NEWS_API_KEY);

  const res = await fetch(url);
  if (!res.ok) return [];
  const data = await res.json();

  return (data.articles || []).map((a) => ({
    title: a.title,
    source: a.source?.name,
    date: a.publishedAt,
    description: a.description,
    url: a.url
  }));
}

export function formatNews(articles) {
  if (!articles.length) return "No recent news available.";
  return articles
    .map((a) => `- [${a.date?.slice(0, 10)}] ${a.title} (${a.source}): ${a.description ?? ""}`)
    .join("\n");
}


===== backend\extract-text.js =====

// Import the inner file: the package root runs a debug routine under ESM and crashes.
import pdf from "pdf-parse/lib/pdf-parse.js";
import mammoth from "mammoth";

// Converts an uploaded file buffer to plain text.
// Scanned PDFs (images only) return almost no text: add OCR (e.g. Tesseract) if you need them.
export async function extractText(file) {
  const name = file.originalname.toLowerCase();

  if (name.endsWith(".pdf")) {
    const data = await pdf(file.buffer);
    return data.text;
  }
  if (name.endsWith(".docx")) {
    const { value } = await mammoth.extractRawText({ buffer: file.buffer });
    return value;
  }
  if (name.endsWith(".txt") || name.endsWith(".csv") || name.endsWith(".md")) {
    return file.buffer.toString("utf-8");
  }
  throw new Error("Unsupported file type. Use PDF, DOCX, TXT or CSV.");
}

    if (!file) return;
    setLoading(true);
    const form = new FormData();
    form.append("file", file);

    const res = await fetch(`${API}/documents`, { method: "POST", body: form });
    const data = await res.json();
    setDoc(data);                               // data.dashboard feeds your charts/KPI cards

    const rep = await fetch(`${API}/documents/${data.id}/reports`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({})
    });
    setReports((await rep.json()).reports);
    setLoading(false);
  }

  async function send() {
    if (!input.trim() || !doc) return;
    const history = messages;
    setMessages([...history, { role: "user", content: input }]);
    setInput("");

    const res = await fetch(`${API}/documents/${doc.id}/chat`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ message: input, history })
    });
    const { answer } = await res.json();
    setMessages((m) => [...m, { role: "assistant", content: answer }]);
  }

  return (
    <div>
      <input type="file" accept=".pdf,.docx,.txt,.csv" onChange={handleUpload} />
      {loading && <p>Analyzing document...</p>}

      {doc && (
        <>
          <h2>{doc.dashboard.document_title}</h2>
          <div>
            {Object.entries(doc.dashboard.departments).map(([dept, d]) => (
              <section key={dept}>
                <h3>{dept}</h3>
                {d.kpis.map((k) => (
                  <div key={k.id}>
                    {k.label}: {k.value} {k.unit}
                    {k.change_pct != null && ` (${k.change_pct}%)`} [{k.status}]
                  </div>
                ))}
                {/* d.charts[i] = { type, title, series } -> feed to Recharts / Chart.js */}
                {d.data_gaps.length > 0 && <small>Missing: {d.data_gaps.join("; ")}</small>}
              </section>
            ))}
          </div>

          {Object.entries(reports).map(([field, text]) => (
            <section key={field}>
              <h3>{field}</h3>
              <pre style={{ whiteSpace: "pre-wrap" }}>{text}</pre>
            </section>
          ))}

          <div>
            {messages.map((m, i) => <p key={i}><b>{m.role}:</b> {m.content}</p>)}
            <input value={input} onChange={(e) => setInput(e.target.value)} onKeyDown={(e) => e.key === "Enter" && send()} />
            <button onClick={send}>Send</button>
          </div>
        </>
      )}
    </div>
  );
}
